// src/domain/services/SubscriptionBillingService.ts
export class SubscriptionBillingService {
  constructor(
    private readonly subscriptionRepo: ISubscriptionRepository,
    private readonly userRepo: IUserRepository,
    private readonly paymentGateway: IPaymentGateway,
    private readonly timeProvider: ITimeProvider
  ) {}

  async processRenewal(userId: string): Promise<RenewalResult> {
    // 1. Fetch through injected ports
    const user = await this.userRepo.getUserById(userId);
    if (!user) return { success: false, message: "User not found" };

    const sub = await this.subscriptionRepo.getSubscriptionByUserId(userId);
    if (!sub) return { success: false, message: "Subscription not found" };

    // 2. Deterministic time injection (No new Date() or temporal coupling)
    const now = this.timeProvider.getCurrentTime();
    const subExpiry = sub.expiresAt instanceof Date ? sub.expiresAt : new Date(sub.expiresAt);
    if (subExpiry.getTime() > now.getTime()) {
      return { success: false, message: "Subscription is not yet expired" };
    }

    // 3. December 10% promotional discount
    let chargeAmount = sub.basePrice;
    if (now.getMonth() === 11) {
      chargeAmount = chargeAmount * 0.9;
    }

    // 4. Charge customer via IPaymentGateway abstraction
    try {
      const chargeSuccess = await this.paymentGateway.charge(user.stripeCustomerId, chargeAmount);
      if (chargeSuccess) {
        const newExpiry = new Date(now.getTime());
        newExpiry.setFullYear(newExpiry.getFullYear() + 1);
        await this.subscriptionRepo.updateExpiration(sub.id, newExpiry);
        return { success: true, message: "Renewal successful" };
      } else {
        return { success: false, message: "Payment failed" };
      }
    } catch (err) {
      return { success: false, message: "Payment gateway error" };
    }
  }
}
// legacy/LegacySubscriptionManager.ts (Original God-Class)
class LegacySubscriptionManager {
  async processRenewal(userId: string) {
    // SMELL 1: Hardcoded DB infrastructure
    const db = new DatabaseConnection(process.env.DB_URL);
    const user = await db.query("SELECT * FROM users WHERE id = ?", [userId]);
    const sub = await db.query("SELECT * FROM subscriptions WHERE user_id = ?", [userId]);

    // SMELL 2: Temporal Coupling (Cannot test December discount deterministically)
    const now = new Date();
    if (new Date(sub.expires_at) > now) {
      return { success: false, message: "Subscription is not yet expired" };
    }

    let chargeAmount = sub.base_price;
    if (now.getMonth() === 11) { chargeAmount = chargeAmount * 0.9; }

    // SMELL 3: Direct Third-Party HTTP client instantiation
    const stripeClient = new ThirdPartyPaymentClient(process.env.PAYMENT_API_KEY);
    const charge = await stripeClient.chargeCustomer(user.stripe_customer_id, chargeAmount);

    // Interleaving SQL updates and responses in a single procedural block
    const newExpiry = new Date(now);
    newExpiry.setFullYear(newExpiry.getFullYear() + 1);
    await db.query("UPDATE subscriptions SET expires_at = ? WHERE id = ?", [newExpiry, sub.id]);
    return { success: true, message: "Renewal successful" };
  }
}
